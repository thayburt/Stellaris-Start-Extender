		}
	}

# Determines if a country needs to spawn additional districts for lithoids or farming
country_event = {
	id = stex_game_start.3
	is_triggered_only = yes
	hide_window = yes

	trigger = { is_country_type = default }
	immediate = {
		if = {
			limit = {
				OR = {
					has_valid_civic = civic_machine_assimilator
					has_valid_civic = civic_machine_servitor
					has_origin = origin_syncretic_evolution
					has_origin = origin_necrophage
				}
			}
			if = {
				limit = { last_created_species = { is_lithoid = yes } }
				set_country_flag = lithoid_subspecies
			}
			else = { set_country_flag = non_lithoid_subspecies }
		}
		else_if = {
			limit = {
				has_origin = origin_tree_of_life
				is_lithoid_empire = yes
			}
			set_country_flag = non_lithoid_subspecies
		}
	}
}

# Lost Colony Origin
country_event = {
	id = stex_game_start_origins.4
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		notai_is_playable_country_trigger = yes
		has_origin = origin_lost_colony
		NOT = { has_country_flag = human_2 }
	}
	immediate = {
		if = {
			limit = {
				NOR = {
					has_global_flag = stex_can_have_lost_colony_ai
					has_global_flag = stex_not_have_lost_colony_ai
				}
			}
			if = {
				limit = {
					OR = {
						any_playable_country = {
							has_origin = origin_lost_colony
							is_ai = no
						}
						check_galaxy_setup_value = { setting = num_advanced_empires value > 0 }
					}
				}
				set_global_flag = stex_can_have_lost_colony_ai
			}
			else = { set_global_flag = stex_not_have_lost_colony_ai }
		}
		if = {
			limit = { has_global_flag = stex_can_have_lost_colony_ai }
			country_event = { id = origin.100 }
		}
		else = {
			log = "REMOVED LOST COLONY"
			set_origin = origin_default
		}
	}
}

	trigger = {
		has_planet_flag = planet_earth

			NOR = {
				has_origin = origin_shattered_ring

		add_deposit = d_mesopotamian_urban_corridor
		add_deposit = d_boswash_metropolitan_axis
		add_deposit = d_mauritanian_security_zone
		add_deposit = d_great_albertan_crater
		add_deposit = d_scandinavian_reclamation_sector
		add_deposit = d_saharan_irrigation_project
		add_deposit = d_pearl_river_agglomerate
		add_deposit = d_great_pacific_garbage_patch
		add_deposit = d_failing_infrastructure_earth

		# Keeping this here just cause it makes more sense for the time being
		if = {
			limit = { owner = { has_valid_civic = civic_machine_servitor } }
			add_deposit = d_hot_springs

	}
