# this = capital planet
# Gatekeeper to manage the populator ordering
planet_event = {
	id = stex_init.4
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		if = {
			limit = { NOT = { has_planet_flag = stex_no_init_planet } }
			clear_deposits = yes
			fire_on_action = {
				on_action = stex_on_capital_init_attempt_special
				scopes = {
					from = from
					fromfrom = fromfrom
				}
			}
			if = {
				limit = { NOT = { stex_planet_is_init_trigger = yes } }
				fire_on_action = {
					on_action = stex_on_capital_init_attempt_origin
					scopes = {
						from = from
						fromfrom = fromfrom
					}
				}
				if = {
					limit = { NOT = { stex_planet_is_init_trigger = yes } }
					fire_on_action = {
						on_action = stex_on_capital_init_attempt_civic
						scopes = {
							from = from
							fromfrom = fromfrom
						}
					}
					if = {
						limit = { NOT = { stex_planet_is_init_trigger = yes } }
						fire_on_action = {
							on_action = stex_on_capital_init_attempt_random
							scopes = {
								from = from
								fromfrom = fromfrom
							}
						}
						if = {
							limit = { stex_planet_is_init_trigger = yes }
							set_planet_flag = stex_planet_init_random
						}
						else = { log_error = "Error occured when trying to init planet [This.GetName] deposits, planet not marked as populated after all passes." }
					}
					else = { set_planet_flag = stex_planet_init_civic }
				}
				else = { set_planet_flag = stex_planet_init_origin }
				fire_on_action = {
					on_action = stex_on_capital_init_always
					scopes = {
						from = from
						fromfrom = fromfrom
					}
				}
			}
			else = { set_planet_flag = stex_planet_init_special }
			stex_unmark_planet_as_init_effect = yes
		}
		if = {
			limit = {
				OR = {
					NOT = { has_planet_flag = stex_no_init_planet }
					AND = {
						has_planet_flag = stex_no_init_planet
						has_planet_flag = stex_populate_colony
					}
				}
			}
			fire_on_action = {
				on_action = stex_on_capital_populate
				scopes = {
					from = from
					fromfrom = fromfrom
				}
			}
		}
	}
}
