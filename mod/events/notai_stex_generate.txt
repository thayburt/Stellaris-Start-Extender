# Used to generate resources for empires
namespace = stex_generate_resources

# Adds starting energy
country_event = {
	id = stex_generate_resources.0
	is_triggered_only = yes
	hide_window = yes

	trigger = { notai_is_playable_country_trigger = yes }
	immediate = { add_resource = { energy = @stex_starting_energy } }
}

# Adds starting consumer goods
country_event = {
	id = stex_generate_resources.4
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		notai_is_playable_country_trigger = yes
		OR = {
			is_gestalt = no
			has_valid_civic = civic_machine_servitor
		}
	}
	immediate = { add_resource = { consumer_goods = @stex_starting_consumer_goods } }
}
